§3 — Conditional Execution Rules (PreExecCheck)
Step 1 — CCBS_SEARCH_SUBSCRIBER_BY_RESOURCE
string-length(//Subscriber/MSISDN/text()) > 0 and not(exists(//Subscriber/SubscriberId))
Only run if MSISDN is present but SubscriberId is not yet populated.
Step 2 — CCBS_GET_ACCOUNT_HEADER
string-length(//Account/AccountID/text()) > 0
Step 3 — CCBS_GET_AGREEMENT_HEADER
exists(//ParentOU/Agreement/AgreementId)
Step 4 — CCBS_GET_SUBS_INFO
boolean(//Subscriber/MSISDN[./text()] and //Subscriber/SubscriberOffers) or boolean(//Subscriber/ResourceRangeInfo)
Step 5 — CCBS_GET_AGREEMENT_INFO
exists(//ParentOU/Agreement) and (//Offers/Action/text() = 'CHG_PARAM' or //Offers/Action/text() = 'ADD')
Step 6 — CCBS_RESOLVE_SOC_CODE
(string-length(//Agreement/Offers/OfferName/text())>0 and //Agreement/Offers/ServiceType/text() != '79')
or (string-length(//SubscriberOffers/OfferName/text())>0 and //SubscriberOffers/ServiceType/text() != '79')
Step 7 — CCBS_GOD
count(//Offers) > 0
or count(//SubscriberOffers) > 0
and (//SubscriberOffers/Action/text() != 'CHG_PARAM' and //ServiceType/text() != '79')
Step 8 — OMX_CAL_OFFERS_EFF_DATE
boolean(//Subscriber/SubscriberOffers[1] and //SubscriberOffers[ExtendedInfo[Name='FE_OR_CCBS' and (Value='FE' or Value='BRMS')]])
Step 9 — OMX_OFFER_INCLUSION
count(//Offers) > 0 or count(//SubscriberOffers) > 0 and (//SubscriberOffers/Action/text() != 'CHG_PARAM' and //Offers/Action/text() != 'CHG_PARAM')
Step 12 — OMX_ADD_FUT_OFFERS
boolean(
  //Subscriber/SubscriberOffers[Action[.='ADD']] and
  //Subscriber/SubscriberOffers[ExtendedInfo[Name='OfferActivityDate' and Value='FUT']][ExtendedInfo[Name='FE_OR_CCBS' and Value='FE']]
  and //SubscriberOffers/ServiceType/text() != '80'
  and //SubscriberOffers/ServiceType/text() != '79'
) or boolean(
  //Agreement/Offers[Action='ADD'] and [...FUT][FE] and ServiceType!='80' and !='79'
)
Step 13 — OMX_ADD_FUT_OFFERS_PRICEPLAN
boolean(
  //Subscriber/SubscriberOffers[OfferActivityDate=FUT][FE] and count(ADD)>0 and ServiceType='80'
)
Step 14 — OMX_ADD_FUT_OFFERS_PARAM
boolean(//SubscriberOffers[FE] and //SubscriberOffers[Action='CHG_PARAM'] and //ParameterInfo[OfferParamActivityDate=FUT])
or boolean(//Agreement/Offers[FE] and [CHG_PARAM] and //ParameterInfo[OfferParamActivityDate=FUT])
Step 15 — CCBS_CHANGE_ACCOUNTING_MANAGEMENT_INFO
string-length(//Account/ExtendedInfo[Name='LEGACY_BAN']/Value/text())!=0 or string-length(//Account/ExtendedInfo[Name='INIT_REASON']/Value/text())!=0 or string-length(//Account/ExtendedInfo[Name='PRIORITY']/Value/text())!=0
Step 16 — CCBS_ADD_PP_OU
exists(//ParentOU/Agreement) and //Offers/ServiceType/text() = '80' and //Offers/ExtendedInfo[Name='FE_OR_CCBS']/Value/text()='FE' and //Offers/Action/text() = 'ADD'
Step 17 — CCBS_CHANGE_PP_OU
exists(//ParentOU/Agreement) and //Offers/ServiceType/text() = '80' and FE_OR_CCBS='FE' and Action='CHG_PP'
Step 18 — CCBS_REMOVE_AGREEMENT_PRICE_PLAN
exists(//ParentOU/Agreement) and //Offers/ServiceType/text() = '80' and FE_OR_CCBS='FE' and Action='REMOVE'
Step 19 — CCBS_UPDATE_AGREEMENT_ON_UNIT
exists(//ParentOU/Agreement) and //Offers/ServiceType/text() != '80' and FE_OR_CCBS='FE'
and (boolean(//Agreement/Offers/ExtendedInfo[Name='OfferActivityDate' and Value='IM'])
     or boolean(//Agreement/Offers/ExtendedInfo[Name='OfferActivityDate' and Value='BD']))
Step 20 — CCBS_CHANGE_PACKAGE_SUBSCRIBER
exists(//ParentOU/Subscriber) and string-length(//SubscriberOffers/Soc)>0 and FE_OR_CCBS='FE' and ServiceType!='79'
and (OfferActivityDate='IM' or ='BD') or ParameterInfo[IM or BD] or Action='CHG_CHARGE'
Step 21 — OMX_ADD_FUT_OFFERS_REMOVE
boolean(//SubscriberOffers[FE] and ExpirationDate and
  ((OfferActivityDate!=FUT and OfferExpActivityDate=FUT and Action=ADD) or (OfferActivityDate=FUT and Action=REMOVE)))
or same for //Agreement/Offers
Step 22 — BL_CREATE_CHARGE
exists(//ParentOU/Subscriber) and FE_OR_CCBS='FE' and ServiceType='79'
Step 23 — CCBS_ADD_RESOURCE_RANGES
count(//ResourceRangeInfo[Action='ADD']) > 0
Step 24 — CCBS_REMOVE_RESOURCE_RANGES
count(//ResourceRangeInfo[Action='REMOVE']) > 0
Step 25 — CCBS_CHANGE_SUBSCRIBER_GENERAL_INFO
//Subscriber/ExtendedInfo[Name='OLD_BAN'|'OLD_BAN_DATE'|'PROJECT_CODE'|'MARKET_CODE'|'SALE_CHANNEL'|'PHASE_CODE']/Value/text()!=''
Step 26 — OMX_UPDATE_FUT_OFFER_COP_DATE
boolean(//SubscriberOffers/ExtendedInfo[Name='FUT_ORDER_ID' or 'FUT_SOC_ID'] and FE and Action='FUT_CHG_DATE')
or boolean(//Agreement/Offers same)
Step 27 — OMX_REMOVE_FUT_OFFER_COP
boolean(//SubscriberOffers/ExtendedInfo[Name='FUT_ORDER_ID' or 'FUT_SOC_ID'] and FE and Action='FUT_REMOVE')
or boolean(//Agreement/Offers same)
Step 29 — OMX_EXP_FUT_REMOVE
exists(//ParentOU/Subscriber) and string-length(//SubscriberOffers/Soc)>0 and FE_OR_CCBS='FE' and Action='REMOVE' and ServiceType!='79'
